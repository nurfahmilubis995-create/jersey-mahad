Panduan Google Apps Script (Code.gs)

Salin kode di bawah ini ke editor Google Apps Script pada Google Sheets Anda. Pastikan folder ID Google Drive sudah diatur ke 1ABRJvupPh4PXy2GdQ47_yr8A9LCS8sGN.

// Google Apps Script (Code.gs)
function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    
    var imageUrl = "";
    if (data.buktiTransfer && data.buktiTransfer.startsWith("data:image")) {
      var folderId = "1ABRJvupPh4PXy2GdQ47_yr8A9LCS8sGN";
      var folder = DriveApp.getFolderById(folderId);
      var blob = Utilities.newBlob(Utilities.base64Decode(data.buktiTransfer.split(",")[1]), data.buktiTransfer.split(";")[0].split(":")[1], "Bukti_" + data.id + ".png");
      var file = folder.createFile(blob);
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      imageUrl = file.getUrl();
    } else {
      imageUrl = data.buktiTransfer || "";
    }
    
    sheet.appendRow([
      data.timestamp,
      data.id,
      data.nama,
      data.angkatan,
      data.lengan,
      data.celana,
      data.ukuran,
      data.namaPunggung,
      data.noPunggung !== null ? data.noPunggung : "Tanpa No",
      data.total,
      imageUrl,
      data.status
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({"status": "success", "id": data.id}))
                        .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": err.toString()}))
                        .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var rows = sheet.getDataRange().getValues();
    var data = [];
    
    for (var i = 1; i < rows.length; i++) {
      var row = rows[i];
      if(!row[1]) continue;
      data.push({
        timestamp: row[0],
        id: row[1],
        nama: row[2],
        angkatan: row[3],
        lengan: row[4],
        celana: row[5],
        ukuran: row[6],
        namaPunggung: row[7],
        noPunggung: row[8] === "Tanpa No" ? null : Number(row[8]),
        total: Number(row[9]),
        buktiTransfer: row[10],
        status: row[11] || "Pending"
      });
    }
    
    return ContentService.createTextOutput(JSON.stringify({"status": "success", "orders": data}))
                        .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": err.toString()}))
                        .setMimeType(ContentService.MimeType.JSON);
  }
}
